const BASE = `https://api.telegram.org/bot${process.env.TELEGRAM_BOT_TOKEN}`; export async function sendMessage(text, opts = {}) { const res = await fetch(`${BASE}/sendMessage`, { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ chat_id: process.env.TELEGRAM_CHAT_ID, text, parse_mode: "HTML", disable_web_page_preview: true, ...opts, }), }); const data = await res.json(); if (!data.ok) throw new Error(`Telegram sendMessage: ${JSON.stringify(data)}`); return data.result; } export async function getUpdates(offset) { const res = await fetch(`${BASE}/getUpdates?timeout=0&offset=${offset}`); const data = await res.json(); if (!data.ok) throw new Error(`Telegram getUpdates: ${JSON.stringify(data)}`); return data.result; } // Baixa um arquivo (ex.: nota de voz .oga) e devolve um Buffer export async function downloadFile(fileId) { const meta = await fetch(`${BASE}/getFile?file_id=${fileId}`).then(r => r.json()); if (!meta.ok) throw new Error(`Telegram getFile: ${JSON.stringify(meta)}`); const url = `https://api.telegram.org/file/bot${process.env.TELEGRAM_BOT_TOKEN}/${meta.result.file_path}`; const buf = await fetch(url).then(r => r.arrayBuffer()); return { buffer: Buffer.from(buf), path: meta.result.file_path }; } // Apaga uma mensagem do chat (usado para limpar os transcripts crus). // O bot só consegue apagar mensagens enviadas por ele mesmo. export async function deleteMessage(messageId) { try { await fetch(`${BASE}/deleteMessage`, { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ chat_id: process.env.TELEGRAM_CHAT_ID, message_id: messageId }), }); } catch { /* silencioso: apagar é best-effort */ } }
